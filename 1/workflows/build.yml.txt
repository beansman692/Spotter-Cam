name: Build Android APK

# This tells GitHub to run the script every time you push code to GitHub
on:
  push:
    branches: [ main, master, dev ]
  pull_request:
    branches: [ main, master ]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    # 1. Downloads a copy of your source code onto the GitHub server
    - name: Checkout project code
      uses: actions/checkout@v4

    # 2. Automatically sets up Java (JDK 17) in the cloud
    - name: Set up JDK 17
      uses: actions/setup-java@v4
      with:
        java-version: '17'
        distribution: 'temurin'
        cache: 'gradle' # Caches dependencies so future builds run much faster

    # 3. Grants the Linux server permission to run your Gradle build wrapper
    - name: Grant execute permission for gradlew
      run: chmod +x gradlew

    # 4. Runs the compilation process to generate the APK file
    - name: Build Debug APK
      run: ./gradlew assembleDebug

    # 5. Grabs the finished APK and uploads it to GitHub so you can download it
    - name: Upload Finished APK
      uses: actions/upload-artifact@v4
      with:
        name: My-Android-App-APK
        path: app/build/outputs/apk/debug/app-debug.apk
